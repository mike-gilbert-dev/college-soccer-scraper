Implementation Notes

URL parameter conventions

Both pages use the same query params as the scoreboard:

?sport=MSO&division=1&season=2025

The [ncaa_team_id] segment is the team's seoname slug (e.g. michigan-st).

Teams list server load (src/routes/teams/+page.server.ts)

import type { PageServerLoad } from './$types';
import { supabaseAdmin } from '$lib/server/supabase-admin';

export const load: PageServerLoad = async ({ url }) => {
  const sportCode  = url.searchParams.get('sport')    ?? 'MSO';
  const division   = parseInt(url.searchParams.get('division') ?? '1', 10);
  const seasonYear = parseInt(url.searchParams.get('season')   ?? '2025', 10);

  // Resolve season id
  const { data: season } = await supabaseAdmin
    .from('seasons')
    .select('id')
    .eq('year', seasonYear)
    .single();

  if (!season) return { teams: [], sportCode, division, seasonYear };

  const { data: teams } = await supabaseAdmin
    .from('team_seasons')
    .select(`
      id,
      division,
      sport_code,
      team:teams ( id, ncaa_team_id, name, short_name ),
      conference:conferences ( name, short_name )
    `)
    .eq('season_id', season.id)
    .eq('division', division)
    .eq('sport_code', sportCode)
    .order('name', { foreignTable: 'teams' });

  return { teams: teams ?? [], sportCode, division, seasonYear };
};

Team detail server load (src/routes/teams/[ncaa_team_id]/+page.server.ts)

import { error } from '@sveltejs/kit';
import type { PageServerLoad } from './$types';
import { supabaseAdmin } from '$lib/server/supabase-admin';

export const load: PageServerLoad = async ({ params, url }) => {
  const sportCode  = url.searchParams.get('sport')    ?? 'MSO';
  const division   = parseInt(url.searchParams.get('division') ?? '1', 10);
  const seasonYear = parseInt(url.searchParams.get('season')   ?? '2025', 10);

  // Load team record
  const { data: team } = await supabaseAdmin
    .from('teams')
    .select('id, ncaa_team_id, name, short_name')
    .eq('ncaa_team_id', params.ncaa_team_id)
    .single();

  if (!team) error(404, 'Team not found');

  // Load team_season for the requested year
  const { data: season } = await supabaseAdmin
    .from('seasons').select('id').eq('year', seasonYear).single();

  let teamSeason = null;
  let players: unknown[] = [];

  if (season) {
    const { data: ts } = await supabaseAdmin
      .from('team_seasons')
      .select('id, division, sport_code, conference:conferences(name)')
      .eq('team_id', team.id)
      .eq('season_id', season.id)
      .single();

    teamSeason = ts;

    if (ts) {
      // Query player_season_stats view joined with players
      const { data: roster } = await supabaseAdmin
        .from('player_season_stats')
        .select('*, player:players(ncaa_player_id, name)')
        .eq('team_season_id', ts.id);

      players = roster ?? [];
    }
  }

  return { team, teamSeason, players, sportCode, division, seasonYear };
};

Roster table sort order

Sort in the Svelte component (client-side) since the view doesn't support ordering easily:

const posOrder: Record<string, number> = { GK: 0, D: 1, M: 2, F: 3 };
const sortedPlayers = $derived(
  [...data.players].sort((a, b) => {
    const pa = posOrder[a.position ?? ''] ?? 9;
    const pb = posOrder[b.position ?? ''] ?? 9;
    if (pa !== pb) return pa - pb;
    return (b.goals ?? 0) - (a.goals ?? 0);
  })
);
