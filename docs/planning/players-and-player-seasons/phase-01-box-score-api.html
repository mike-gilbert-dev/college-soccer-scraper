Endpoint details

The box score uses the same NCAA APQ pattern as existing endpoints. Full URL (ready to test):

https://sdataprod.ncaa.com/?meta=NCAA_GetGamecenterBoxscoreSoccerById_web&extensions={"persistedQuery":{"version":1,"sha256Hash":"c9070c4e5a76468a4025896df89f8a7b22be8275c54a22ff79619cbb27d63d7d"}}&variables={"contestId":"6465168","staticTestEnv":null}

Hash: c9070c4e5a76468a4025896df89f8a7b22be8275c54a22ff79619cbb27d63d7d
Variable: contestId (string, not integer) — matches games.ncaa_contest_id in the DB.
Always pass staticTestEnv: null.

New test endpoint: src/routes/api/scrape/test-boxscore/+server.ts

Follow the same pattern as the existing /api/scrape/test/+server.ts. Auth-gate with safeGetSession().

import { json, error } from '@sveltejs/kit';
import type { RequestHandler } from './$types';
import { fetchRawBoxScore } from '$lib/server/ncaa-api';

export const GET: RequestHandler = async ({ url, locals }) => {
  const { user } = await locals.safeGetSession();
  if (!user) error(401, 'Unauthorized');

  const contestId = url.searchParams.get('contestId');
  if (!contestId) error(400, 'contestId is required');

  const data = await fetchRawBoxScore(contestId);
  return json(data);
};

export async function fetchRawBoxScore(contestId: string): Promise<unknown> {
  const url = new URL(GQL_HOST);
  url.searchParams.set('meta', 'NCAA_GetGamecenterBoxscoreSoccerById_web');
  url.searchParams.set('extensions', JSON.stringify({
    persistedQuery: { version: 1, sha256Hash: BOX_SCORE_HASH }
  }));
  url.searchParams.set('variables', JSON.stringify({ contestId, staticTestEnv: null }));

  const res = await fetch(url.toString(), {
    headers: {
      'Accept': 'application/json',
      'User-Agent': 'Mozilla/5.0 (compatible; college-soccer-scraper/1.0)'
    }
  });
  if (!res.ok) throw new Error(`NCAA box score API ${res.status} for contestId=${contestId}`);
  return res.json();
}

export async function fetchBoxScore(contestId: string): Promise<NcaaBoxScore> {
  const raw = await fetchRawBoxScore(contestId) as { data?: { contestByContestId?: NcaaBoxScore } };
  // CONFIRM this path matches the actual API response before proceeding to Phase 3
  const boxScore = raw?.data?.contestByContestId;
  if (!boxScore) throw new Error(`No box score data for contestId=${contestId}`);
  return boxScore;
}
